-----------------------------------------------------------------------
first-party/applications/experimentRunner/launcher.py:340-349, originating at 273.
relay.start() runs before the new handler is installed. An ordinary Ctrl+C there
can raise KeyboardInterrupt through the existing caller handler. The enclosing
finally restores signals/closes the relay BEFORE the except BaseException at 348
records the primary error in pendingExit. If relay.close() raises an ordinary
RuntimeError, that error replaces the primary interrupt. launcherFailure observes
only the cleanup exception and execute returns an ordinary failed envelope instead
of re-raising the original KeyboardInterrupt/attaching its diagnostic snapshot.
This does not produce CLI success, but violates the explicit first-interrupt and
independent cleanup-failure contract. It requires only ONE interrupt followed by
an ordinary cleanup error, not repeated interruption inside provider shutdown.

synthetic and no model is launched (startup occurs before first unit). This test
combines two paths each covered separately by existing worker tests; those separate

def test_review_ex03_first_start_interrupt_survives_relay_close_failure(rig, monkeypatch):
    """Ordinary relay cleanup failure must not swallow the primary startup interrupt."""
    originalStart = launcher._SignalCancellationRelay.start
    originalClose = launcher._SignalCancellationRelay.close
    first = KeyboardInterrupt('primary-start-interrupt')
    relays = []
    def start(self):
        """Start the exact owned helper, then simulate Ctrl+C before handler installation."""
        originalStart(self)
        relays.append(self)
        raise first
    def close(self):
        """Join the helper normally, then report an independent ordinary cleanup failure."""
        originalClose(self)
        raise RuntimeError('secondary-relay-close-failure')
    monkeypatch.setattr(launcher._SignalCancellationRelay, 'start', start)
    monkeypatch.setattr(launcher._SignalCancellationRelay, 'close', close)
    with pytest.raises(KeyboardInterrupt) as caught:
        launcher.execute(launcher.prepare(definition(), rig[1]), rig[2])
    assert caught.value is first
    assert relays and not relays[0]._thread.is_alive()
    assert caught.value.experimentExecution.value()['hostCleanup'] == 'completed'
